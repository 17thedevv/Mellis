enum Option[T] {
    Some(T),
    None
}

fn unwrap(value: Option[i32]) -> i32 {
    match value {
        Option::Some(v) => v,
        Option::None => 0,
    }
}
fn add_all(values: [i32]) -> i32 {
    let total = 0;
    for value in values {
        total += value;
    }
    total
}
